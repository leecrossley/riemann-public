\newcommand*{\sbtv}[2]{\csname sbtw#1@#2\endcsname}
\newcommand*{\sbtwrow}[1]{%
  \foreach \c in {q,g,a,b,Nid,Ninv,bound,total,lower,weil}{%
    \pgfplotstablegetelem{#1}{\c}\of{\sbtwists}%
    \global\expandafter\let\csname sbtw#1@\c\endcsname\pgfplotsretval}%
  \pgfmathsetmacro{\sbtwtmp}{\sbtv{#1}{total}/2}%
  \global\expandafter\let\csname sbtw#1@mid\endcsname\sbtwtmp
  \pgfmathsetmacro{\sbtwtmp}{\sbtv{#1}{total}-\sbtv{#1}{weil}}%
  \global\expandafter\let\csname sbtw#1@weillow\endcsname\sbtwtmp
  \pgfmathsetmacro{\sbtwtmp}{1.16*\sbtv{#1}{bound}}%
  \global\expandafter\let\csname sbtw#1@ymax\endcsname\sbtwtmp}
\sbtwrow{0}\sbtwrow{1}\sbtwrow{5}
\newcommand*{\sbtwpanel}[1]{%
  \nextgroupplot[title={$q=\sbtv{#1}{q}$, $g=\sbtv{#1}{g}$,
      $(a,b)=(\sbtv{#1}{a},\sbtv{#1}{b})$}, ymin=0, ymax=\sbtv{#1}{ymax}]
  \fill[sbWeil!15] (axis cs:0.35,\sbtv{#1}{weillow}) rectangle
    (axis cs:2.65,\sbtv{#1}{weil});
  \draw[sb weil line] (axis cs:0.35,\sbtv{#1}{weillow}) --
    (axis cs:2.65,\sbtv{#1}{weillow})
    (axis cs:0.35,\sbtv{#1}{weil}) -- (axis cs:2.65,\sbtv{#1}{weil});
  \addplot[ybar, bar width=0.5, fill=sbZero!45, draw=sbZero,
    nodes near coords, every node near coord/.append style={font=\scriptsize,
    text=sbZero!70!black}]
    coordinates {(1,\sbtv{#1}{Nid}) (2,\sbtv{#1}{Ninv})};
  \draw[sbOther, line width=0.7pt, densely dotted]
    (axis cs:0.35,\sbtv{#1}{mid}) -- (axis cs:2.65,\sbtv{#1}{mid});
  \draw[sb bound line, line width=1.1pt] (axis cs:0.35,\sbtv{#1}{bound}) --
    (axis cs:2.65,\sbtv{#1}{bound})
    node[pos=1, anchor=south east, font=\scriptsize, text=sbInk,
      inner sep=1pt] {$\sbtv{#1}{bound}$};
  \draw[sbBound, line width=0.9pt] (axis cs:0.35,\sbtv{#1}{lower}) --
    (axis cs:2.65,\sbtv{#1}{lower})
    node[pos=1, anchor=south east, font=\scriptsize, text=sbInk,
      inner sep=1pt] {$\sbtv{#1}{lower}$};
  \draw[sbBound, ->, line width=0.6pt] (axis cs:0.5,\sbtv{#1}{bound}) --
    (axis cs:0.5,\sbtv{#1}{lower});
}
